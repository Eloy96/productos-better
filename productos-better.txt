  <script>
    const DEFAULT_API_URL = "https://script.google.com/macros/s/AKfycbwczNs-lS1aUyJxpvSMuYGekx9zSHxNwWONxw3PdkLhTIbG3D6LqE7SLx4mgl-SwGV5/exec";
    const API_URL_KEY = "productos_better_admin_api_url_simple";

    const state = {
      apiUrl: localStorage.getItem(API_URL_KEY) || DEFAULT_API_URL,
      products: [],
      filtered: [],
      mode: "create",
      currentPhoto: null,
      removeImage: false,
      stream: null,
      qr: null,
      busy: false
    };

    const $ = id => document.getElementById(id);

    document.addEventListener("DOMContentLoaded", () => {
      bindEvents();
      $("api-url-input").value = state.apiUrl;
      loadInventory(true);
      lucide.createIcons();
    });

    function bindEvents() {
      $("search-button").addEventListener("click", checkSku);
      $("new-button").addEventListener("click", newProductFromInput);
      $("sku-input").addEventListener("keydown", event => { if (event.key === "Enter") checkSku(); });
      $("filter-input").addEventListener("input", renderInventory);
      $("refresh-button").addEventListener("click", () => loadInventory(true));
      $("close-editor").addEventListener("click", closeEditor);
      $("save-button").addEventListener("click", saveProduct);
      $("add-one-button").addEventListener("click", addOne);
      $("delete-button").addEventListener("click", deleteProduct);
      $("photo-button").addEventListener("click", () => startCamera("photo"));
      $("clear-photo").addEventListener("click", clearPhoto);
      $("scan-button").addEventListener("click", () => startCamera("scan"));
      $("close-camera").addEventListener("click", stopCamera);
      $("capture-button").addEventListener("click", capturePhoto);
      $("settings-button").addEventListener("click", openSettings);
      $("close-settings").addEventListener("click", closeSettings);
      $("save-settings").addEventListener("click", saveSettings);
    }

    async function loadInventory(showLoader) {
      if (showLoader) $("loading-state").classList.remove("hidden");

      try {
        const response = await fetch(`${state.apiUrl}?action=getAll&limit=1000`, { redirect: "follow" });
        const json = await response.json();
        if (json.status !== "success") throw new Error(json.error || "Respuesta invalida");

        state.products = (json.data || []).map(normalizeProduct).filter(p => p.sku);
        renderInventory();
        $("status-label").textContent = "Actualizado " + new Date().toLocaleTimeString("es-MX", { hour: "2-digit", minute: "2-digit" });
      } catch (error) {
        console.error(error);
        $("status-label").textContent = "Sin conexion";
        toast("No se pudo cargar el inventario", "error");
      } finally {
        $("loading-state").classList.add("hidden");
      }
    }

    function renderInventory() {
      const terms = normalizeText($("filter-input").value).split(" ").filter(Boolean);
      state.filtered = state.products.filter(product => {
        const haystack = normalizeText(`${product.name} ${product.sku}`);
        return terms.every(term => haystack.includes(term));
      });

      const list = $("inventory-list");
      list.innerHTML = "";

      if (state.filtered.length === 0) {
        list.innerHTML = '<div class="rounded-xl border border-slate-200 bg-white p-8 text-center text-slate-500 md:col-span-2">Sin productos</div>';
      } else {
        const fragment = document.createDocumentFragment();
        state.filtered.forEach(product => fragment.appendChild(inventoryCard(product)));
        list.appendChild(fragment);
      }

      renderStats();
      lucide.createIcons();
    }

    function inventoryCard(product) {
      const card = document.createElement("button");
      card.type = "button";
      card.className = "rounded-xl bg-white border border-slate-200 p-3 shadow-sm text-left flex items-center gap-3 active:scale-[0.99]";
      card.addEventListener("click", () => openEditor("update", product));

      const imgBox = document.createElement("div");
      imgBox.className = "w-16 h-16 rounded-lg bg-slate-50 border border-slate-100 overflow-hidden grid place-items-center shrink-0";
      if (product.imageSafe) {
        const img = document.createElement("img");
        img.src = product.imageSafe;
        img.alt = product.name;
        img.className = "w-full h-full object-contain";
        img.onerror = () => img.replaceWith(imagePlaceholder());
        imgBox.appendChild(img);
      } else {
        imgBox.appendChild(imagePlaceholder());
      }

      const info = document.createElement("div");
      info.className = "min-w-0 flex-1";
      info.innerHTML = `
        <h3 class="font-black text-sm leading-snug line-clamp-2">${escapeHtml(product.name)}</h3>
        <p class="font-mono text-[11px] text-slate-500 mt-1">${escapeHtml(product.sku)}</p>
        <p class="font-black text-emerald-700">${formatPrice(product.price)}</p>
      `;

      const stock = document.createElement("div");
      stock.className = "text-right shrink-0";
      stock.innerHTML = `<p class="text-2xl font-black ${product.stock <= 0 ? "text-red-600" : "text-blue-700"}">${product.stock}</p><p class="text-[10px] font-black uppercase text-slate-400">stock</p>`;

      card.append(imgBox, info, stock);
      return card;
    }

    async function checkSku() {
      const sku = normalizeSku($("sku-input").value);
      if (!sku) return toast("Ingresa un SKU", "error");

      setBusy($("search-button"), true, "Buscando...");
      try {
        const response = await fetch(`${state.apiUrl}?sku=${encodeURIComponent(sku)}`, { redirect: "follow" });
        const json = await response.json();
        if (json.exists) openEditor("update", normalizeProduct(json.data));
        else openEditor("create", emptyProduct(sku));
      } catch (error) {
        console.error(error);
        toast("No se pudo buscar", "error");
      } finally {
        setBusy($("search-button"), false);
      }
    }

    function newProductFromInput() {
      const sku = normalizeSku($("sku-input").value);
      if (!sku) return toast("Escribe o escanea un SKU", "error");
      const existing = state.products.find(p => p.sku === sku);
      openEditor(existing ? "update" : "create", existing || emptyProduct(sku));
    }

    function openEditor(mode, product) {
      state.mode = mode;
      state.currentPhoto = null;
      state.removeImage = false;

      $("editor-section").classList.remove("hidden");
      $("editor-title").textContent = mode === "create" ? "Nuevo producto" : "Editar producto";
      $("editor-subtitle").textContent = product.sku || "";
      $("form-sku").value = product.sku || "";
      $("form-sku").disabled = mode === "update";
      $("form-name").value = product.name || "";
      $("form-stock").value = Number(product.stock || 0);
      $("form-price").value = Number(product.price || 0);
      $("existing-actions").classList.toggle("hidden", mode !== "update");

      if (product.imageSafe) {
        $("photo-preview").src = product.imageSafe;
        $("photo-wrap").classList.remove("hidden");
        $("photo-button").classList.add("hidden");
      } else {
        resetPhotoUI();
      }

      $("editor-section").scrollIntoView({ behavior: "smooth", block: "start" });
      lucide.createIcons();
    }

    function closeEditor() {
      $("editor-section").classList.add("hidden");
      state.currentPhoto = null;
      state.removeImage = false;
    }

    async function saveProduct() {
      const sku = normalizeSku($("form-sku").value);
      const name = $("form-name").value.trim();
      const stock = parseInt($("form-stock").value, 10);
      const price = Number($("form-price").value);

      if (!sku) return toast("Falta SKU", "error");
      if (!name) return toast("Falta nombre", "error");
      if (!Number.isFinite(stock) || stock < 0) return toast("Stock invalido", "error");
      if (!Number.isFinite(price) || price < 0) return toast("Precio invalido", "error");

      const payload = {
        action: state.mode === "create" ? "create" : "update",
        sku,
        name,
        stock,
        price,
        imageBase64: state.currentPhoto,
        removeImage: state.removeImage
      };

      setBusy($("save-button"), true, "Guardando...");
      try {
        const json = await post(payload);
        if (!["success", "updated"].includes(json.status)) throw new Error(json.error || "No se pudo guardar");
        toast(state.mode === "create" ? "Producto creado" : "Producto actualizado");
        closeEditor();
        await loadInventory(false);
      } catch (error) {
        console.error(error);
        toast(error.message || "No se pudo guardar", "error");
      } finally {
        setBusy($("save-button"), false);
      }
    }

    async function addOne() {
      const sku = normalizeSku($("form-sku").value);
      setBusy($("add-one-button"), true, "Sumando...");
      try {
        const json = await post({ action: "add", sku, stock: 1 });
        if (json.status !== "updated") throw new Error(json.error || "No se pudo sumar");
        toast("Stock +1");
        closeEditor();
        await loadInventory(false);
      } catch (error) {
        console.error(error);
        toast(error.message || "No se pudo sumar", "error");
      } finally {
        setBusy($("add-one-button"), false);
      }
    }

    async function deleteProduct() {
      const sku = normalizeSku($("form-sku").value);
      const name = $("form-name").value.trim();
      if (!confirm(`Eliminar "${name}"?\nSKU: ${sku}`)) return;

      setBusy($("delete-button"), true, "Eliminando...");
      try {
        const json = await post({ action: "delete", sku });
        if (json.status !== "deleted") throw new Error(json.error || "No se pudo eliminar");
        toast("Producto eliminado");
        closeEditor();
        await loadInventory(false);
      } catch (error) {
        console.error(error);
        toast(error.message || "No se pudo eliminar", "error");
      } finally {
        setBusy($("delete-button"), false);
      }
    }

    async function post(payload) {
      const response = await fetch(state.apiUrl, {
        method: "POST",
        redirect: "follow",
        headers: { "Content-Type": "text/plain;charset=utf-8" },
        body: JSON.stringify(payload)
      });
      return response.json();
    }

    async function startCamera(mode) {
      document.body.classList.add("camera-open");

      if (mode === "scan") {
        $("camera-video").classList.add("hidden");
        $("photo-controls").classList.add("hidden");
        $("qr-reader").classList.remove("hidden");

        if (!window.Html5Qrcode) {
          toast("No cargo el escaner", "error");
          stopCamera();
          return;
        }

        state.qr = new Html5Qrcode("qr-reader");
        try {
          await state.qr.start({ facingMode: "environment" }, { fps: 10, qrbox: { width: 250, height: 250 } }, text => {
            $("sku-input").value = normalizeSku(text);
            stopCamera();
            checkSku();
          }, () => {});
        } catch (error) {
          console.error(error);
          toast("No se pudo abrir camara", "error");
          stopCamera();
        }
        return;
      }

      $("qr-reader").classList.add("hidden");
      $("camera-video").classList.remove("hidden");
      $("photo-controls").classList.remove("hidden");

      try {
        state.stream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: "environment", width: { ideal: 1920 }, height: { ideal: 1080 } }
        });
        $("camera-video").srcObject = state.stream;
      } catch (error) {
        console.error(error);
        toast("No se pudo abrir camara", "error");
        stopCamera();
      }
    }

    function stopCamera() {
      document.body.classList.remove("camera-open");
      if (state.stream) {
        state.stream.getTracks().forEach(track => track.stop());
        state.stream = null;
      }
      if (state.qr) {
        state.qr.stop().then(() => state.qr.clear()).catch(() => {});
        state.qr = null;
      }
    }

    function capturePhoto() {
      const video = $("camera-video");
      if (!video.videoWidth || !video.videoHeight) return toast("Camara cargando", "error");

      const canvas = $("photo-canvas");
      const maxSide = 1400;
      const scale = Math.min(1, maxSide / Math.max(video.videoWidth, video.videoHeight));
      canvas.width = Math.round(video.videoWidth * scale);
      canvas.height = Math.round(video.videoHeight * scale);
      canvas.getContext("2d").drawImage(video, 0, 0, canvas.width, canvas.height);

      state.currentPhoto = canvas.toDataURL("image/jpeg", 0.82);
      state.removeImage = false;
      $("photo-preview").src = state.currentPhoto;
      $("photo-wrap").classList.remove("hidden");
      $("photo-button").classList.add("hidden");
      stopCamera();
    }

    function clearPhoto() {
      state.currentPhoto = null;
      state.removeImage = true;
      resetPhotoUI();
    }

    function resetPhotoUI() {
      $("photo-preview").src = "";
      $("photo-wrap").classList.add("hidden");
      $("photo-button").classList.remove("hidden");
    }

    function openSettings() {
      $("api-url-input").value = state.apiUrl;
      document.body.classList.add("settings-open");
    }

    function closeSettings() {
      document.body.classList.remove("settings-open");
    }

    function saveSettings() {
      const url = $("api-url-input").value.trim();
      if (!url.startsWith("https://script.google.com/")) return toast("URL invalida", "error");
      state.apiUrl = url;
      localStorage.setItem(API_URL_KEY, url);
      closeSettings();
      toast("Configuracion guardada");
      loadInventory(true);
    }

    function renderStats() {
      $("stat-products").textContent = state.products.length;
      $("stat-stock").textContent = state.products.reduce((sum, p) => sum + Math.max(0, p.stock), 0);
      $("stat-empty").textContent = state.products.filter(p => p.stock <= 0).length;
    }

    function normalizeProduct(product) {
      return {
        name: String(product.name || "").trim(),
        sku: normalizeSku(product.sku),
        stock: Number(product.stock || 0),
        price: Number(product.price || 0),
        image: String(product.image || ""),
        imageSafe: imageUrl(product.image)
      };
    }

    function emptyProduct(sku) {
      return { sku, name: "", stock: 1, price: 0, image: "", imageSafe: "" };
    }

    function imageUrl(link) {
      const value = String(link || "").trim();
      if (!value) return "";
      const match = value.match(/id=([a-zA-Z0-9_-]{20,})/) || value.match(/\/d\/([a-zA-Z0-9_-]{20,})\//);
      if (match) return `https://drive.google.com/thumbnail?id=${match[1]}&sz=w800`;
      return value;
    }

    function imagePlaceholder() {
      const div = document.createElement("div");
      div.className = "w-full h-full bg-slate-100 text-slate-300 grid place-items-center";
      div.innerHTML = '<i data-lucide="image-off" class="w-6 h-6"></i>';
      return div;
    }

    function setBusy(button, busy, label) {
      if (!button) return;
      button.disabled = busy;
      if (busy) {
        button.dataset.html = button.innerHTML;
        button.innerHTML = `<div class="loader !w-5 !h-5 !border-2"></div>${label ? `<span>${label}</span>` : ""}`;
      } else if (button.dataset.html) {
        button.innerHTML = button.dataset.html;
      }
      lucide.createIcons();
    }

    function normalizeSku(value) {
      return String(value || "").trim().toUpperCase();
    }

    function normalizeText(value) {
      return String(value || "").normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase();
    }

    function formatPrice(value) {
      return Number(value || 0).toLocaleString("es-MX", { style: "currency", currency: "MXN" });
    }

    function escapeHtml(value) {
      return String(value || "").replace(/[&<>"']/g, char => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#039;" }[char]));
    }

    function toast(message, type = "success") {
      const el = $("toast");
      el.textContent = message;
      el.className = `fixed top-24 left-1/2 -translate-x-1/2 z-[80] max-w-[92vw] rounded-full px-5 py-3 text-sm font-bold text-white shadow-2xl text-center ${type === "error" ? "bg-red-600" : "bg-emerald-600"}`;
      el.classList.remove("hidden");
      clearTimeout(toast.timer);
      toast.timer = setTimeout(() => el.classList.add("hidden"), 3000);
    }
